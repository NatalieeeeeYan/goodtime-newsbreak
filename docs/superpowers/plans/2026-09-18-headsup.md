# Heads Up Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Python service that polls local-event sources, decides whether each event affects the user's day, and pushes an actionable card to Telegram with one-tap actions and follow-up chat.

**Architecture:** Two plug-in layers (world-side `Sensors`, user-side `Integrations`) feed an agent core of `prefilter → LLM relevance → LLM planner → channel push`. All layers talk only through Pydantic models in `headsup/models.py` and `typing.Protocol`s in `headsup/integrations/protocols.py`. Providers are chosen in `integrations.yaml`.

**Tech Stack:** Python 3.12, uv, pydantic 2, pydantic-ai-slim[openai] (DeepSeek via OpenAI-compatible provider), python-telegram-bot 21+, APScheduler 3, httpx, feedparser, PyYAML, python-dotenv, google-api-python-client + google-auth-oauthlib, pytest + pytest-asyncio.

**Spec:** `docs/superpowers/specs/2026-09-16-headsup-design.md`

## Global Constraints

- Python `>=3.12`; manage with `uv`. Run everything via `uv run ...`.
- No ORM: SQLite through `sqlite3` stdlib.
- No comments in code (project rule). Names carry the meaning.
- Every provider wrapper ≤ 60 lines, every planner tool ≤ 30 lines; providers return only models from `headsup/models.py`.
- One shipped implementation per Protocol (+ Mock where the demo needs it). Do not build a second real provider.
- Never push on LLM failure or `confidence < 0.6`. Push rate cap default 3/hour. `DEMO_MODE=1` allows only sources in `{"nws", "inject"}` to reach the channel.
- Scenario JSON times are relative (`"+45m"`, `"tomorrow 09:00"`) and resolved at inject time.
- Telegram `callback_data` is limited to 64 bytes: it is `"{card_id}:{action.id}"` where `card_id` is the SQLite row id and `action.id` is `a1`/`a2`/`a3`.
- LLM tests never hit the network: `pydantic_ai.models.ALLOW_MODEL_REQUESTS = False` in `tests/conftest.py`; use `TestModel(custom_output_args=...)` / `FunctionModel`.
- Tests are marked `llm` only when they need a real model; `uv run pytest -m "not llm"` must stay green offline.
- Commit trailer on every commit: `Co-Authored-By: Claude Opus 5 <noreply@anthropic.com>`.

**Deviations from the spec, decided here:**
- Eventbrite's public event search endpoint no longer exists; the events sensor uses **Ticketmaster Discovery API** (`TICKETMASTER_KEY`).
- The planner **proposes** actions; side-effecting calls (`calendar_insert`, `schedule_reminder`) run only when the user taps the button (`headsup/actions.py`). Read-only tools (`route_eta`, `find_places`, `calendar_list`, `free_slots`, `checkout_url`) are planner tools.
- `LocalEvent.extra: dict` carries source-specific facts (e.g. `eta_delay_minutes` for an injected crash) so injected scenarios produce realistic numbers when live traffic is calm.
- Local-news RSS sensor is the last task and optional; it geo-tags events at the profile's home point instead of LLM location extraction.

```python
from __future__ import annotations

import re
from datetime import datetime, timedelta, timezone

from headsup.models import CalendarEvent, FreeSlot

_OFFSET = re.compile(r"^([+-])((?:\d+[mhd])+)$")
_PART = re.compile(r"(\d+)([mhd])")
_DAY_TIME = re.compile(r"^(today|tomorrow)\s+(\d{1,2}):(\d{2})$")
_UNIT = {"m": "minutes", "h": "hours", "d": "days"}


def utcnow() -> datetime:
    return datetime.now(timezone.utc)


def parse_relative(value: str | datetime, now: datetime) -> datetime:
    if isinstance(value, datetime):
        return value
    text = value.strip()
    if m := _OFFSET.match(text):
        sign, body = m.groups()
        delta = sum((timedelta(**{_UNIT[u]: int(n)}) for n, u in _PART.findall(body)), timedelta())
        return now + delta if sign == "+" else now - delta
    if m := _DAY_TIME.match(text):
        day, hh, mm = m.groups()
        base = now + timedelta(days=1) if day == "tomorrow" else now
        return base.replace(hour=int(hh), minute=int(mm), second=0, microsecond=0)
    parsed = datetime.fromisoformat(text)
    return parsed if parsed.tzinfo else parsed.replace(tzinfo=timezone.utc)


def compute_free_slots(
    events: list[CalendarEvent], start: datetime, end: datetime, min_minutes: int
) -> list[FreeSlot]:
    cursor = start
    slots: list[FreeSlot] = []
    for ev in sorted(events, key=lambda e: e.start):
        if ev.start > cursor and (ev.start - cursor) >= timedelta(minutes=min_minutes):
            slots.append(FreeSlot(start=cursor, end=min(ev.start, end)))
        cursor = max(cursor, ev.end)
        if cursor >= end:
            break
    if cursor < end and (end - cursor) >= timedelta(minutes=min_minutes):
        slots.append(FreeSlot(start=cursor, end=end))
    return slots
```

```python
from __future__ import annotations

import math

from headsup.models import GeoPoint

EARTH_KM = 6371.0


def haversine_km(a: GeoPoint, b: GeoPoint) -> float:
    la1, lo1, la2, lo2 = map(math.radians, (a.lat, a.lon, b.lat, b.lon))
    h = math.sin((la2 - la1) / 2) ** 2 + math.cos(la1) * math.cos(la2) * math.sin((lo2 - lo1) / 2) ** 2
    return 2 * EARTH_KM * math.asin(math.sqrt(h))


def _xy(p: GeoPoint, ref_lat: float) -> tuple[float, float]:
    return (math.radians(p.lon) * math.cos(math.radians(ref_lat)) * EARTH_KM, math.radians(p.lat) * EARTH_KM)


def distance_to_segment_km(p: GeoPoint, a: GeoPoint, b: GeoPoint) -> float:
    ref = (a.lat + b.lat) / 2
    px, py = _xy(p, ref)
    ax, ay = _xy(a, ref)
    bx, by = _xy(b, ref)
    dx, dy = bx - ax, by - ay
    if dx == 0 and dy == 0:
        return haversine_km(p, a)
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / (dx * dx + dy * dy)))
    cx, cy = ax + t * dx, ay + t * dy
    return math.hypot(px - cx, py - cy)
```

```python
@dataclass
class Candidate:
    event: LocalEvent
    matched_events: list[CalendarEvent] = field(default_factory=list)
    free_slots: list[FreeSlot] = field(default_factory=list)
    reason: str = ""


def _window(event: LocalEvent) -> tuple[datetime, datetime]:
    return event.starts_at, event.ends_at or event.starts_at + DEFAULT_DURATION


def _in_time(event: LocalEvent, now: datetime) -> bool:
    start, end = _window(event)
    return start <= now + LOOKAHEAD and end >= now


def _geo_reason(event: LocalEvent, profile: Profile, calendar_events: list[CalendarEvent]) -> str | None:
    p = event.point
    for anchor in profile.anchors():
        if haversine_km(p, anchor) <= event.radius_km:
            return f"within {event.radius_km:.0f} km of {anchor.label or 'anchor'}"
    for ce in calendar_events:
        if ce.geo and haversine_km(p, ce.geo) <= event.radius_km:
            return f"near calendar event '{ce.title}'"
    if event.type == "traffic":
        width = max(event.radius_km, CORRIDOR_MIN_KM)
        for dest in (profile.work, profile.school):
            if dest and distance_to_segment_km(p, profile.home, dest) <= width:
                return f"on commute corridor home→{dest.label or 'destination'}"
    return None


def _matched(event: LocalEvent, calendar_events: list[CalendarEvent]) -> list[CalendarEvent]:
    start, end = _window(event)
    return [ce for ce in calendar_events if ce.end >= start - CALENDAR_PAD and ce.start <= end + CALENDAR_PAD]


def _overlapping_slots(event: LocalEvent, free_slots: list[FreeSlot]) -> list[FreeSlot]:
    start, end = _window(event)
    out = []
    for s in free_slots:
        lo, hi = max(s.start, start), min(s.end, end)
        if hi - lo >= FREE_SLOT_MIN:
            out.append(s)
    return out


def prefilter(
    event: LocalEvent,
    profile: Profile,
    calendar_events: list[CalendarEvent],
    free_slots: list[FreeSlot],
    now: datetime,
) -> Candidate | None:
    if not _in_time(event, now):
        return None
    reason = _geo_reason(event, profile, calendar_events)
    if reason is None:
        return None
    slots = _overlapping_slots(event, free_slots)
    if event.type == "event" and not slots:
        return None
    return Candidate(event=event, matched_events=_matched(event, calendar_events), free_slots=slots, reason=reason)
```

```python
from __future__ import annotations

import sqlite3
from datetime import datetime
from pathlib import Path

from headsup.models import Card

SCHEMA = """
CREATE TABLE IF NOT EXISTS seen_events (event_id TEXT PRIMARY KEY, seen_at TEXT NOT NULL);
CREATE TABLE IF NOT EXISTS cards (id INTEGER PRIMARY KEY AUTOINCREMENT, chat_id INTEGER NOT NULL, event_id TEXT NOT NULL, json TEXT NOT NULL, created_at TEXT NOT NULL);
CREATE TABLE IF NOT EXISTS channel_messages (chat_id INTEGER NOT NULL, message_id INTEGER NOT NULL, card_id INTEGER NOT NULL, PRIMARY KEY (chat_id, message_id));
CREATE TABLE IF NOT EXISTS threads (id INTEGER PRIMARY KEY AUTOINCREMENT, card_id INTEGER NOT NULL, role TEXT NOT NULL, text TEXT NOT NULL);
CREATE TABLE IF NOT EXISTS pushes (event_id TEXT NOT NULL, pushed_at TEXT NOT NULL);
"""


class Storage:
    def __init__(self, path: Path | str):
        self.conn = sqlite3.connect(str(path), check_same_thread=False)
        self.conn.executescript(SCHEMA)

    def seen(self, event_id: str) -> bool:
        row = self.conn.execute("SELECT 1 FROM seen_events WHERE event_id=?", (event_id,)).fetchone()
        return row is not None

    def mark_seen(self, event_id: str, at: datetime) -> None:
        with self.conn:
            self.conn.execute("INSERT OR IGNORE INTO seen_events VALUES (?, ?)", (event_id, at.isoformat()))

    def save_card(self, chat_id: int, card: Card) -> int:
        with self.conn:
            cur = self.conn.execute(
                "INSERT INTO cards (chat_id, event_id, json, created_at) VALUES (?, ?, ?, ?)",
                (chat_id, card.event_id, card.model_dump_json(), datetime.now().isoformat()),
            )
        return int(cur.lastrowid)

    def get_card(self, card_id: int) -> Card | None:
        row = self.conn.execute("SELECT json FROM cards WHERE id=?", (card_id,)).fetchone()
        return Card.model_validate_json(row[0]) if row else None

    def map_message(self, chat_id: int, message_id: int, card_id: int) -> None:
        with self.conn:
            self.conn.execute("INSERT OR REPLACE INTO channel_messages VALUES (?, ?, ?)", (chat_id, message_id, card_id))

    def card_for_message(self, chat_id: int, message_id: int) -> tuple[int, Card] | None:
        row = self.conn.execute(
            "SELECT card_id FROM channel_messages WHERE chat_id=? AND message_id=?", (chat_id, message_id)
        ).fetchone()
        if not row:
            return None
        card = self.get_card(int(row[0]))
        return (int(row[0]), card) if card else None

    def latest_card_for_chat(self, chat_id: int) -> tuple[int, Card] | None:
        row = self.conn.execute(
            "SELECT id, json FROM cards WHERE chat_id=? ORDER BY id DESC LIMIT 1", (chat_id,)
        ).fetchone()
        return (int(row[0]), Card.model_validate_json(row[1])) if row else None

    def append_thread(self, card_id: int, role: str, text: str) -> None:
        with self.conn:
            self.conn.execute("INSERT INTO threads (card_id, role, text) VALUES (?, ?, ?)", (card_id, role, text))

    def get_thread(self, card_id: int) -> list[tuple[str, str]]:
        rows = self.conn.execute("SELECT role, text FROM threads WHERE card_id=? ORDER BY id", (card_id,)).fetchall()
        return [(r[0], r[1]) for r in rows]

    def record_push(self, event_id: str, at: datetime) -> None:
        with self.conn:
            self.conn.execute("INSERT INTO pushes VALUES (?, ?)", (event_id, at.isoformat()))

    def pushes_since(self, at: datetime) -> int:
        row = self.conn.execute("SELECT COUNT(*) FROM pushes WHERE pushed_at >= ?", (at.isoformat(),)).fetchone()
        return int(row[0])
```

```python
from __future__ import annotations

import json
from collections.abc import Callable
from datetime import datetime
from pathlib import Path

from headsup.models import LocalEvent
from headsup.timeutil import parse_relative, utcnow


class InjectorSensor:
    name = "inject"

    def __init__(self, scenarios_dir: Path, now_fn: Callable[[], datetime] = utcnow):
        self.dir = scenarios_dir
        self.now_fn = now_fn
        self.queue: list[str] = []

    def list_scenarios(self) -> list[str]:
        return sorted(p.stem for p in self.dir.glob("*.json"))

    def load(self, name: str) -> LocalEvent:
        now = self.now_fn()
        raw = json.loads((self.dir / f"{name}.json").read_text())
        raw["starts_at"] = parse_relative(raw["starts_at"], now)
        raw["ends_at"] = parse_relative(raw["ends_at"], now) if raw.get("ends_at") else None
        raw["id"] = f"{raw['id']}:{int(now.timestamp())}"
        return LocalEvent.model_validate(raw)

    def enqueue(self, name: str) -> None:
        self.queue.append(name)

    async def poll(self) -> list[LocalEvent]:
        names, self.queue = self.queue, []
        return [self.load(n) for n in names]
```

The timestamp suffix on `id` lets the same scenario be replayed in rehearsal without tripping dedupe.

```python
from pydantic_ai import Agent
from pydantic_ai.models import Model

from headsup.llm import MODEL_SETTINGS
from headsup.models import Impact

MIN_CONFIDENCE = 0.6

RELEVANCE_INSTRUCTIONS = """You decide whether a local event affects one specific person's next 24 hours.
You are given the event, the person's profile, the calendar entries that overlap the event window, and why a rule-based filter flagged it.
Answer strictly about THIS person. affected=true only if the event plausibly changes what they should do, when they should leave, or where they should be.
`why` is one sentence addressed to the person ("Your 9:00 school run uses I-580 W, which is now blocked."). Name the calendar entry when one is hit and put its id in calendar_event_id.
severity: high = must act within hours; medium = should adjust plans today; low = nice to know.
confidence is your honest probability that affected is correct."""

relevance_agent: Agent[None, Impact] = Agent(output_type=Impact, instructions=RELEVANCE_INSTRUCTIONS)


def _fmt_calendar(events: list[CalendarEvent]) -> str:
    if not events:
        return "(none)"
    return "\n".join(f"- [{e.id}] {e.title}: {e.start.isoformat()} → {e.end.isoformat()} @ {e.location or e.geo and e.geo.label or '?'}" for e in events)

    ev = candidate.event
    return f"""Now: {now.isoformat()}
Person: {profile.name}; home {profile.home.label}; work {profile.work.label if profile.work else '-'}; school {profile.school.label if profile.school else '-'}; usual commute starts {profile.commute_start}; preferences: {', '.join(profile.preferences) or '-'}

Event ({ev.type}, severity {ev.severity}/5, source {ev.source}):
{ev.title}
{ev.summary}
Window: {ev.starts_at.isoformat()} → {ev.ends_at.isoformat() if ev.ends_at else 'open'}; affects ~{ev.radius_km:.0f} km around ({ev.lat}, {ev.lon})
Extra facts: {ev.extra or '-'}

Why flagged: {candidate.reason}

Calendar entries overlapping the window:
{_fmt_calendar(candidate.matched_events)}

Free slots overlapping the window: {len(candidate.free_slots)}"""


async def judge(candidate: Candidate, profile: Profile, model: Model, now: datetime) -> Impact:
    result = await relevance_agent.run(build_relevance_prompt(candidate, profile, now), model=model, model_settings=MODEL_SETTINGS)
```

```python
from __future__ import annotations

from datetime import datetime, timedelta, timezone

import httpx

from headsup.models import GeoPoint, Place, RouteEta

ROUTES_URL = "https://routes.googleapis.com/directions/v2:computeRoutes"
PLACES_URL = "https://places.googleapis.com/v1/places:searchText"
GEOCODE_URL = "https://maps.googleapis.com/maps/api/geocode/json"
ROUTE_MASK = "routes.duration,routes.distanceMeters,routes.description"
PLACE_MASK = "places.displayName,places.formattedAddress,places.location,places.currentOpeningHours.openNow,places.rating,places.googleMapsUri"


class GoogleMaps:
    def __init__(self, api_key: str, transport: httpx.AsyncBaseTransport | None = None):
        self.key = api_key
        self.client = httpx.AsyncClient(transport=transport, timeout=10)

    async def route_eta(self, origin: str, destination: str, depart_at: datetime, traffic_aware: bool = True) -> RouteEta:
        depart = max(depart_at, datetime.now(timezone.utc) + timedelta(minutes=1))
        body = {
            "origin": {"address": origin}, "destination": {"address": destination}, "travelMode": "DRIVE",
            "routingPreference": "TRAFFIC_AWARE_OPTIMAL" if traffic_aware else "TRAFFIC_UNAWARE",
            "departureTime": depart.isoformat().replace("+00:00", "Z"),
        }
        r = await self.client.post(ROUTES_URL, json=body, headers={"X-Goog-Api-Key": self.key, "X-Goog-FieldMask": ROUTE_MASK})
        r.raise_for_status()
        route = r.json()["routes"][0]
        seconds = int(route["duration"].rstrip("s"))
        return RouteEta(origin=origin, destination=destination, minutes=round(seconds / 60), distance_km=round(route["distanceMeters"] / 1000, 1), summary=route.get("description", ""))

    async def find_places(self, query: str, near: GeoPoint, open_now: bool = True, limit: int = 3) -> list[Place]:
        body = {
            "textQuery": query, "openNow": open_now, "maxResultCount": limit,
            "locationBias": {"circle": {"center": {"latitude": near.lat, "longitude": near.lon}, "radius": 3000.0}},
        }
        r = await self.client.post(PLACES_URL, json=body, headers={"X-Goog-Api-Key": self.key, "X-Goog-FieldMask": PLACE_MASK})
        r.raise_for_status()
        return [
            Place(
                name=p["displayName"]["text"], address=p.get("formattedAddress", ""),
                lat=p["location"]["latitude"], lon=p["location"]["longitude"],
                open_now=p.get("currentOpeningHours", {}).get("openNow"), rating=p.get("rating"), maps_url=p["googleMapsUri"],
            )
            for p in r.json().get("places", [])[:limit]
        ]

    async def geocode(self, text: str) -> GeoPoint:
        r = await self.client.get(GEOCODE_URL, params={"address": text, "key": self.key})
        r.raise_for_status()
        first = r.json()["results"][0]
        loc = first["geometry"]["location"]
        return GeoPoint(lat=loc["lat"], lon=loc["lng"], label=first["formatted_address"])
```

```python
from __future__ import annotations

from datetime import datetime
from uuid import uuid4

from apscheduler.schedulers.asyncio import AsyncIOScheduler

from headsup.integrations.protocols import Channel


class SchedulerReminders:
    def __init__(self, scheduler: AsyncIOScheduler, channel: Channel):
        self.scheduler = scheduler
        self.channel = channel

    def schedule(self, at: datetime, chat_id: int, text: str) -> str:
        rid = f"rem:{uuid4().hex[:8]}"
        self.scheduler.add_job(self.channel.send_text, "date", run_date=at, args=[chat_id, f"⏰ {text}"], id=rid)
        return rid

    def cancel(self, reminder_id: str) -> None:
        self.scheduler.remove_job(reminder_id)
```

```python
from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timedelta

from pydantic_ai import RunContext
from pydantic_ai.toolsets import FunctionToolset

from headsup.integrations.protocols import Booking, Calendar, Maps
from headsup.models import CalendarEvent, FreeSlot, GeoPoint, Place, Profile, RouteEta


@dataclass
class AgentDeps:
    profile: Profile
    calendar: Calendar
    maps: Maps
    booking: Booking
    now: datetime


def resolve_place(profile: Profile, text: str) -> str:
    named = {"home": profile.home, "work": profile.work, "school": profile.school}
    p = named.get(text.strip().lower())
    return f"{p.lat},{p.lon}" if p else text


def resolve_point(profile: Profile, text: str) -> GeoPoint:
    named = {"home": profile.home, "work": profile.work, "school": profile.school}
    return named.get(text.strip().lower()) or profile.home


toolset: FunctionToolset[AgentDeps] = FunctionToolset()


@toolset.tool
async def route_eta(ctx: RunContext[AgentDeps], origin: str, destination: str, depart_in_minutes: int = 0, traffic_aware: bool = True) -> RouteEta:
    """Driving time between two places. origin/destination accept the labels home, work, school or any address. traffic_aware=False gives the typical no-traffic baseline."""
    p = ctx.deps.profile
    depart = ctx.deps.now + timedelta(minutes=depart_in_minutes)
    return await ctx.deps.maps.route_eta(resolve_place(p, origin), resolve_place(p, destination), depart, traffic_aware)


@toolset.tool
async def find_places(ctx: RunContext[AgentDeps], query: str, near: str = "home", open_now: bool = True) -> list[Place]:
    """Search places (e.g. 'cafe with wifi', 'indoor playground') near home, work, school or an address. Returns up to 3 with a Google Maps link."""
    point = resolve_point(ctx.deps.profile, near)
    if point is ctx.deps.profile.home and near.strip().lower() not in {"home", ""}:
        point = await ctx.deps.maps.geocode(near)
    return await ctx.deps.maps.find_places(query, point, open_now=open_now, limit=3)


@toolset.tool
def calendar_list(ctx: RunContext[AgentDeps], hours_ahead: int = 24) -> list[CalendarEvent]:
    """The person's calendar entries from now until hours_ahead."""
    return ctx.deps.calendar.list_events(ctx.deps.now, ctx.deps.now + timedelta(hours=hours_ahead))


@toolset.tool
def free_slots(ctx: RunContext[AgentDeps], hours_ahead: int = 48, min_minutes: int = 120) -> list[FreeSlot]:
    """Gaps of at least min_minutes in the person's calendar within hours_ahead."""
    return ctx.deps.calendar.free_slots(ctx.deps.now, ctx.deps.now + timedelta(hours=hours_ahead), min_minutes)


@toolset.tool
def checkout_url(ctx: RunContext[AgentDeps], url: str | None, title: str) -> str:
    """A link where the person can buy tickets / book for the event. Pass the event url if known."""
    return ctx.deps.booking.checkout_url(url, title)
```

```python
from __future__ import annotations

from datetime import datetime

from pydantic_ai import Agent
from pydantic_ai.models import Model

from headsup.llm import MODEL_SETTINGS
from headsup.models import Action, Card, Impact, LocalEvent
from headsup.relevance import Candidate
from headsup.tools import AgentDeps, toolset

PLANNER_INSTRUCTIONS = """You are Heads Up, a local-life assistant. A local event has been judged to affect this person. Produce ONE card that tells them what it means and gives 1-3 concrete actions.

Rules:
- Ground every number in a tool result. Use route_eta (traffic_aware=False for the usual baseline) when travel is involved; add any eta_delay_minutes from the event's extra facts on top of the baseline and say so. Use find_places for alternatives (cafes with wifi, indoor venues). Use free_slots / calendar_list before proposing to add something.
- headline: ≤ 60 chars, names the event and the person's plan it hits.
- impact: 1-2 sentences, second person, specific times.
- detail: optional supporting facts (ETA comparison, top places with rating).
- actions: 1-3, most useful first. Button labels ≤ 40 chars. Kinds and payloads:
  remind      {"at": ISO-8601 datetime, "text": what to say}
  calendar_add {"title", "start": ISO, "end": ISO, "location"}
  navigate    {"name", "maps_url"}   (maps_url from find_places)
  open_url    {"url"}                (tickets/info; use checkout_url)
  chat        {"prompt"}             (a follow-up question you would answer well)
- Never invent places, links or times. If a tool fails, drop that action."""

planner_agent: Agent[AgentDeps, Card] = Agent(output_type=Card, deps_type=AgentDeps, instructions=PLANNER_INSTRUCTIONS, toolsets=[toolset])


def build_planner_prompt(event: LocalEvent, impact: Impact, candidate: Candidate, now: datetime) -> str:
    cal = "\n".join(f"- [{e.id}] {e.title}: {e.start.isoformat()} → {e.end.isoformat()} @ {e.location or '-'}" for e in candidate.matched_events) or "(none)"
    return f"""Now: {now.isoformat()}

Event ({event.type}, severity {event.severity}/5): {event.title}
{event.summary}
Window: {event.starts_at.isoformat()} → {event.ends_at.isoformat() if event.ends_at else 'open'}
Extra facts: {event.extra or '-'}
Source: {event.url or event.source}

Judged impact ({impact.severity}, confidence {impact.confidence:.2f}): {impact.why}
Hit calendar entry: {impact.calendar_event_id or '-'}

Calendar entries in the window:
{cal}

Free slots in the window: {[(s.start.isoformat(), s.end.isoformat()) for s in candidate.free_slots] or '-'}"""


def normalise(card: Card, event: LocalEvent) -> Card:
    actions = [Action(id=f"a{i + 1}", label=a.label[:40], kind=a.kind, payload=a.payload) for i, a in enumerate(card.actions[:3])]
    return Card(event_id=event.id, headline=card.headline, impact=card.impact, detail=card.detail, actions=actions)


async def plan(event: LocalEvent, impact: Impact, candidate: Candidate, deps: AgentDeps, model: Model) -> Card:
    result = await planner_agent.run(build_planner_prompt(event, impact, candidate, deps.now), deps=deps, model=model, model_settings=MODEL_SETTINGS)
    return normalise(result.output, event)
```

```python
from __future__ import annotations

import html
from dataclasses import dataclass
from datetime import datetime

from headsup.integrations.protocols import Booking, Calendar, Reminders
from headsup.models import Action


@dataclass
class ActionDeps:
    calendar: Calendar
    reminders: Reminders
    booking: Booking
    chat_id: int


def _dt(value: str) -> datetime:
    return datetime.fromisoformat(value)


async def execute_action(action: Action, deps: ActionDeps) -> str:
    p = action.payload
    if action.kind == "remind":
        at = _dt(p["at"])
        deps.reminders.schedule(at, deps.chat_id, p["text"])
        return f"✅ Reminder set for {at.strftime('%H:%M')}: {html.escape(p['text'])}"
    if action.kind == "calendar_add":
        ev = deps.calendar.insert_event(p["title"], _dt(p["start"]), _dt(p["end"]), p.get("location"))
        return f"✅ Added <b>{html.escape(ev.title)}</b> {ev.start.strftime('%a %H:%M')}–{ev.end.strftime('%H:%M')}"
    if action.kind == "navigate":
        return f'🧭 <a href="{html.escape(p["maps_url"])}">Navigate to {html.escape(p["name"])}</a>'
    if action.kind == "open_url":
        url = deps.booking.checkout_url(p.get("url"), action.label)
        return f'🔗 <a href="{html.escape(url)}">{html.escape(action.label)}</a>'
    if action.kind == "chat":
        return p["prompt"]
    raise ValueError(action.kind)
```

```python
from __future__ import annotations

import asyncio
import html
from functools import partial

from telegram import InlineKeyboardButton, InlineKeyboardMarkup, Update
from telegram.constants import ParseMode
from telegram.ext import Application, CallbackQueryHandler, CommandHandler, ContextTypes, MessageHandler, filters

from headsup.integrations.protocols import CallbackHandler, CommandHandler as CommandFn, ReplyHandler
from headsup.models import Card
from headsup.storage import Storage


def render_card(card: Card) -> str:
    parts = [f"<b>{html.escape(card.headline)}</b>", html.escape(card.impact)]
    if card.detail:
        parts.append(f"<i>{html.escape(card.detail)}</i>")
    return "\n\n".join(parts)


def keyboard(card: Card, card_id: int) -> InlineKeyboardMarkup:
    buttons = [InlineKeyboardButton(a.label, callback_data=f"{card_id}:{a.id}") for a in card.actions]
    rows = [buttons[:1], buttons[1:]] if len(buttons) > 1 else [buttons]
    return InlineKeyboardMarkup([r for r in rows if r])


def parse_callback(data: str) -> tuple[int, str]:
    card_id, action_id = data.split(":", 1)
    return int(card_id), action_id


class TelegramChannel:
    def __init__(self, token: str, storage: Storage):
        self.storage = storage
        self.app = Application.builder().token(token).build()
        self._callback: CallbackHandler | None = None
        self._reply: ReplyHandler | None = None
        self.app.add_handler(CallbackQueryHandler(self._on_callback_query))
        self.app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, self._on_text))

    async def push(self, chat_id: int, card: Card) -> int:
        card_id = self.storage.save_card(chat_id, card)
        msg = await self.app.bot.send_message(chat_id, render_card(card), parse_mode=ParseMode.HTML, reply_markup=keyboard(card, card_id), disable_web_page_preview=True)
        self.storage.map_message(chat_id, msg.message_id, card_id)
        return msg.message_id

    async def edit(self, chat_id: int, message_id: int, text: str) -> None:
        found = self.storage.card_for_message(chat_id, message_id)
        if not found:
            await self.app.bot.edit_message_text(text, chat_id=chat_id, message_id=message_id, parse_mode=ParseMode.HTML)
            return
        card_id, card = found
        await self.app.bot.edit_message_text(render_card(card) + "\n\n" + text, chat_id=chat_id, message_id=message_id, parse_mode=ParseMode.HTML, reply_markup=keyboard(card, card_id), disable_web_page_preview=True)

    async def send_text(self, chat_id: int, text: str) -> int:
        msg = await self.app.bot.send_message(chat_id, text, parse_mode=ParseMode.HTML, disable_web_page_preview=True)
        return msg.message_id

    def on_callback(self, fn: CallbackHandler) -> None:
        self._callback = fn

    def on_reply(self, fn: ReplyHandler) -> None:
        self._reply = fn

    def on_command(self, name: str, fn: CommandFn) -> None:
        self.app.add_handler(CommandHandler(name, partial(self._on_command, fn)))

    async def _on_command(self, fn: CommandFn, update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
        await fn(update.effective_chat.id, list(context.args or []))

    async def _on_callback_query(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
        q = update.callback_query
        await q.answer()
        if self._callback and q.message and q.data:
            await self._callback(q.message.chat_id, q.message.message_id, q.data)

    async def _on_text(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
        if self._reply and update.message and update.message.text:
            await self._reply(update.effective_chat.id, update.message.text)

    async def run(self) -> None:
        async with self.app:
            await self.app.start()
            await self.app.updater.start_polling()
            try:
                await asyncio.Event().wait()
            finally:
                await self.app.updater.stop()
                await self.app.stop()
```

```python
from __future__ import annotations

import logging
from collections.abc import Callable, Iterable
from datetime import datetime, timedelta

from pydantic_ai.models import Model

from headsup.config import Settings
from headsup.integrations.registry import Integrations
from headsup.models import Card, LocalEvent
from headsup.planner import plan
from headsup.relevance import MIN_CONFIDENCE, judge, prefilter
from headsup.sensors import Sensor
from headsup.storage import Storage
from headsup.timeutil import utcnow
from headsup.tools import AgentDeps

log = logging.getLogger("headsup.pipeline")


class Pipeline:
    def __init__(self, settings: Settings, storage: Storage, integrations: Integrations, model: Model | None, now_fn: Callable[[], datetime] = utcnow):
        self.settings = settings
        self.storage = storage
        self.integrations = integrations
        self.model = model
        self.now_fn = now_fn

    async def handle(self, event: LocalEvent) -> Card | None:
        now = self.now_fn()
        if self.storage.seen(event.id):
            return None
        self.storage.mark_seen(event.id, now)
        if self.settings.demo_mode and event.source not in self.settings.demo_allowed_sources:
            log.info("demo mode: muted %s from %s", event.id, event.source)
            return None
        profile = self.integrations.profile.get()
        if profile.telegram_chat_id is None:
            log.warning("no chat bound; send /start to the bot")
            return None
        if self.storage.pushes_since(now - timedelta(hours=1)) >= self.settings.push_rate_cap:
            log.warning("rate cap hit; dropping %s", event.id)
            return None
        cal = self.integrations.calendar
        events = cal.list_events(now, now + timedelta(hours=24))
        free = cal.free_slots(now, now + timedelta(hours=48), 120)
        cand = prefilter(event, profile, events, free, now)
        if cand is None:
            log.info("prefilter dropped %s", event.id)
            return None
        log.info("candidate %s: %s", event.id, cand.reason)
        try:
            impact = await judge(cand, profile, self.model, now)
        except Exception:
            log.exception("relevance failed for %s", event.id)
            return None
        log.info("impact %s: affected=%s conf=%.2f why=%s", event.id, impact.affected, impact.confidence, impact.why)
        if not impact.affected or impact.confidence < MIN_CONFIDENCE:
            return None
        deps = AgentDeps(profile=profile, calendar=cal, maps=self.integrations.maps, booking=self.integrations.booking, now=now)
        try:
            card = await plan(event, impact, cand, deps, self.model)
        except Exception:
            log.exception("planner failed for %s", event.id)
            return None
        await self.integrations.channel.push(profile.telegram_chat_id, card)
        self.storage.record_push(event.id, now)
        log.info("pushed card for %s: %s", event.id, card.headline)
        return card

    async def poll(self, sensors: Iterable[Sensor]) -> None:
        for sensor in sensors:
            try:
                events = await sensor.poll()
            except Exception:
                log.exception("sensor %s failed", sensor.name)
                continue
            for event in events:
                await self.handle(event)
```

```python
    ch = app.integrations.channel
    integ = app.integrations

    def agent_deps() -> AgentDeps:
        return AgentDeps(profile=integ.profile.get(), calendar=integ.calendar, maps=integ.maps, booking=integ.booking, now=utcnow())

    async def on_start(chat_id: int, args: list[str]) -> None:
        integ.profile.update({"telegram_chat_id": chat_id})
        await ch.send_text(chat_id, "👋 Heads Up is watching your area. I'll message you when something nearby affects your day.\nTry: /demo 580_accident")

        await ch.send_text(chat_id, f"sensors: {', '.join(s.name for s in app.sensors)}\ndemo_mode: {app.settings.demo_mode}\nmodel: {app.settings.deepseek_model}")

        latest = app.storage.latest_card_for_chat(chat_id)
        if latest is None:
            await ch.send_text(chat_id, "Nothing to discuss yet. Try /demo 580_accident")
            return
        card_id, card = latest
        thread = app.storage.get_thread(card_id)
        try:
            answer = await reply(text, card, thread, agent_deps(), app.model)
        except Exception:
            log.exception("chat failed")
            await ch.send_text(chat_id, "Sorry, I couldn't work that out just now.")
            return
        app.storage.append_thread(card_id, "user", text)
        app.storage.append_thread(card_id, "assistant", answer)
        await ch.send_text(chat_id, answer)

    async def on_callback(chat_id: int, message_id: int, data: str) -> None:
        card_id, action_id = parse_callback(data)
        card = app.storage.get_card(card_id)
        action = next((a for a in card.actions if a.id == action_id), None) if card else None
        if action is None:
            return

            await on_reply(chat_id, action.payload["prompt"])
            return

        except Exception:

            result = "⚠️ That didn't work."
        await ch.edit(chat_id, message_id, result)

    ch.on_command("start", on_start)
    ch.on_command("demo", on_demo)
    ch.on_command("status", on_status)
    ch.on_reply(on_reply)
    ch.on_callback(on_callback)


async def main() -> None:
    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(name)s %(levelname)s %(message)s")
    app = build_app()
    wire_handlers(app)
    app.scheduler.add_job(app.pipeline.poll, "interval", minutes=app.settings.poll_interval_minutes, args=[app.sensors], id="poll")
    app.scheduler.start()
    log.info("sensors: %s", [s.name for s in app.sensors])
    await app.integrations.channel.run()


if __name__ == "__main__":
    asyncio.run(main())
```

```python
from datetime import datetime, timedelta, timezone

def _dt(value: str | None) -> datetime | None:
    return datetime.fromisoformat(value) if value else None


class NwsSensor:
    name = "nws"

    def __init__(self, home: GeoPoint, transport: httpx.AsyncBaseTransport | None = None):
        self.home = home
        self.client = httpx.AsyncClient(transport=transport, timeout=10, headers={"User-Agent": "headsup (hackathon demo)", "Accept": "application/geo+json"})

    async def poll(self) -> list[LocalEvent]:
        r = await self.client.get(URL, params={"point": f"{self.home.lat},{self.home.lon}"})
        r.raise_for_status()
        out = []
        for f in r.json().get("features", []):
            p = f["properties"]
            start = _dt(p.get("onset") or p.get("effective")) or datetime.now(timezone.utc)
            end = _dt(p.get("ends") or p.get("expires")) or start + timedelta(hours=6)
            out.append(LocalEvent(
                id=f"nws:{p['id']}", source="nws", type="weather", title=p.get("headline") or p["event"],
                summary=(p.get("description") or "")[:600], lat=self.home.lat, lon=self.home.lon, radius_km=30.0,
                starts_at=start, ends_at=end, severity=SEVERITY.get(p.get("severity", "Unknown"), 1), url=p.get("web"),
                extra={"event": p.get("event")},
            ))
```

```python
from __future__ import annotations

from datetime import datetime, timezone
from pathlib import Path

from google.auth.transport.requests import Request
from google.oauth2.credentials import Credentials
from google_auth_oauthlib.flow import InstalledAppFlow
from googleapiclient.discovery import build

from headsup.models import CalendarEvent, FreeSlot
from headsup.timeutil import compute_free_slots

SCOPES = ["https://www.googleapis.com/auth/calendar"]


def parse_item(item: dict) -> CalendarEvent | None:
    start, end = item.get("start", {}).get("dateTime"), item.get("end", {}).get("dateTime")
    if not start or not end:
        return None
    return CalendarEvent(
        id=item["id"], title=item.get("summary", "(untitled)"),
        start=datetime.fromisoformat(start).astimezone(timezone.utc), end=datetime.fromisoformat(end).astimezone(timezone.utc),
        location=item.get("location"),
    )


class GoogleCalendar:
    def __init__(self, client_json: Path, token_path: Path | None = None):
        self.client_json = client_json
        self.token_path = token_path or client_json.parent / "token.json"
        self._service = None

    def _creds(self) -> Credentials:
        creds = Credentials.from_authorized_user_file(str(self.token_path), SCOPES) if self.token_path.exists() else None
        if creds and creds.expired and creds.refresh_token:
            creds.refresh(Request())
        elif not creds or not creds.valid:
            creds = InstalledAppFlow.from_client_secrets_file(str(self.client_json), SCOPES).run_local_server(port=0)
        self.token_path.write_text(creds.to_json())
        return creds

    def service(self):
        if self._service is None:
            self._service = build("calendar", "v3", credentials=self._creds(), cache_discovery=False)
        return self._service

    def list_events(self, start: datetime, end: datetime) -> list[CalendarEvent]:
        res = self.service().events().list(calendarId="primary", timeMin=start.isoformat(), timeMax=end.isoformat(), singleEvents=True, orderBy="startTime", maxResults=50).execute()
        return [ev for ev in (parse_item(i) for i in res.get("items", [])) if ev]

    def free_slots(self, start: datetime, end: datetime, min_minutes: int) -> list[FreeSlot]:
        return compute_free_slots(self.list_events(start, end), start, end, min_minutes)

    def insert_event(self, title: str, start: datetime, end: datetime, location: str | None = None) -> CalendarEvent:
        body = {"summary": title, "location": location, "start": {"dateTime": start.isoformat()}, "end": {"dateTime": end.isoformat()}}
        created = self.service().events().insert(calendarId="primary", body=body).execute()
        return parse_item(created) or CalendarEvent(id=created["id"], title=title, start=start, end=end, location=location)
```
